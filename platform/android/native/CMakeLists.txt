cmake_minimum_required(VERSION 3.22)
project(CrystalCoreAndroid C CXX)

# CrystalCore's native libraries for Android arm64: the translation (src/gen) and Rhipper's
# runtime32-posix, as libCrystalCore.so, and the JNI boot library libCrystalCore_boot.so that loads it
# below 2 GB. Built with Rhipper's toolchain (stock clang >= 22, the NDK's sysroot):
#
#   cmake -G Ninja -B build/android -S platform/android/native \
#     -DCMAKE_TOOLCHAIN_FILE=<rhipper>/toolchains/android-arm64-llvm.cmake \
#     -DANDROID_NDK=<sdk>/ndk/28.2.13676358 -DCMAKE_BUILD_TYPE=Release
#   cmake --build build/android -j 2
#
# The libraries land in platform/android/app/src/main/jniLibs/arm64-v8a for the Gradle build.

set(CC_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/../../..)
get_filename_component(CC_ROOT ${CC_ROOT} ABSOLUTE)
file(READ ${CC_ROOT}/rhipper.json RHIPPER_JSON)
string(JSON RHIPPER_DIR GET ${RHIPPER_JSON} rhipper)
set(RUNTIME32_POSIX ${RHIPPER_DIR}/runtime32-posix)

# ---- the translation, as this build compiles it -----------------------------------------------------
# src/gen is regenerated, never edited: the build works on a copy of it with Rhipper's current
# Ghidra32.h and translate32's port_longs (the reference's 32-bit `long`, spelled gen_long). Once
# src/gen is regenerated by a translate32 that has both, the copy is the same as the original.
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(GEN_PORT ${CMAKE_BINARY_DIR}/gen)
file(GLOB GEN_ORIGINAL CONFIGURE_DEPENDS ${CC_ROOT}/src/gen/*.c ${CC_ROOT}/src/gen/*.h)
set(GEN_SOURCES "")
foreach(f ${GEN_ORIGINAL})
  get_filename_component(n ${f} NAME)
  if(n MATCHES "\\.c$")
    list(APPEND GEN_SOURCES ${GEN_PORT}/${n})
  endif()
endforeach()
add_custom_command(
  OUTPUT ${GEN_PORT}/.ported
  COMMAND ${CMAKE_COMMAND} -E remove_directory ${GEN_PORT}
  COMMAND ${CMAKE_COMMAND} -E copy_directory ${CC_ROOT}/src/gen ${GEN_PORT}
  COMMAND ${CMAKE_COMMAND} -E copy ${RHIPPER_DIR}/runtime32/Ghidra32.h ${GEN_PORT}/Ghidra32.h
  COMMAND ${Python3_EXECUTABLE} ${RHIPPER_DIR}/translate32.py --port-longs ${GEN_PORT}
  COMMAND ${CMAKE_COMMAND} -E touch ${GEN_PORT}/.ported
  WORKING_DIRECTORY ${CC_ROOT}
  DEPENDS ${GEN_ORIGINAL} ${RHIPPER_DIR}/runtime32/Ghidra32.h ${RHIPPER_DIR}/translate32.py
  BYPRODUCTS ${GEN_SOURCES}
  COMMENT "src/gen -> ${GEN_PORT} (Ghidra32.h, port_longs)")
add_custom_target(gen_port DEPENDS ${GEN_PORT}/.ported)

include(${RUNTIME32_POSIX}/Runtime32Posix.cmake)
add_library(gen OBJECT ${GEN_SOURCES})
add_dependencies(gen gen_port)
target_include_directories(gen PRIVATE ${GEN_PORT})
rhipper_gen_posix(gen)
target_compile_options(gen PRIVATE -O2 -g -w)

# ---- the libraries ----------------------------------------------------------------------------------
add_library(crystalcore SHARED $<TARGET_OBJECTS:gen>)
rhipper_runtime32_posix(crystalcore NAME CrystalCore ENV CRYSTALCORE REF ffxiv.exe)

set(JNILIBS ${CC_ROOT}/platform/android/app/src/main/jniLibs/arm64-v8a)
set_target_properties(crystalcore crystalcore_boot PROPERTIES LIBRARY_OUTPUT_DIRECTORY ${JNILIBS})
